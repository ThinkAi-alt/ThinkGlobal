// Trip Estimator
document.getElementById("estimator-form").addEventListener("submit", function(e) {
  e.preventDefault();
  const days = parseInt(document.getElementById("days").value);
  const budget = parseInt(document.getElementById("budget").value);
  const destination = document.getElementById("destination").value;

  if (!days || !destination) {
    document.getElementById("estimate-result").innerText = "Please enter all fields.";
    return;
  }

  const total = days * budget;
  document.getElementById("estimate-result").innerText =
    `Estimated cost for ${days} days in ${destination}: $${total}`;

  // Load hotels and food recommendations dynamically
  loadHotels(destination);
  loadFood(destination);
});
